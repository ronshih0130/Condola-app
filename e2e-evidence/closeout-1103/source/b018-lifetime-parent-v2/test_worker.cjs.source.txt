'use strict';
const {test}=require('node:test'),assert=require('node:assert/strict'),fs=require('node:fs'),os=require('node:os'),path=require('node:path'),crypto=require('node:crypto');
const {parseStrict,exactKeys,checkedFile,verifyApplication,validateClosure}=require('./phase-worker.cjs');
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
test('strict JSON rejects nested duplicate aliases, invalid UTF-8 and unsupported numbers',()=>{for(const b of [Buffer.from('{"x":{"a":1,"\\u0061":2}}'),Buffer.from([123,34,120,34,58,34,255,34,125]),Buffer.from('{"x":1.5}'),Buffer.from('{"x":1e3}'),Buffer.from('{"x":9007199254740992}'),Buffer.from('{"x":"\\ud800"}')])assert.throws(()=>parseStrict(b));assert.deepEqual(parseStrict(Buffer.from('{"x":[1,true,null]}')),{x:[1,true,null]});});
test('exact key sets cannot collide through a delimiter',()=>{assert.throws(()=>exactKeys({'a\0b':1,c:2},['a','b\0c']));exactKeys({a:1,b:2},['b','a']);});
test('omitted or changed transitive source is rejected before any import',()=>{const self='/self',fixture='/fixture',checks='/checks',integrity='/integrity',a='a'.repeat(64),b='b'.repeat(64),known={[fixture]:b,[checks]:b,[integrity]:b},locked={[self]:a,...known},pins={...locked},required=[self,fixture,checks,integrity];assert.equal(validateClosure(required,pins,locked,known,self),true);const omitted={...pins};delete omitted[checks];assert.throws(()=>validateClosure(required,omitted,locked,known,self));assert.throws(()=>validateClosure(required,{...pins,[integrity]:a},locked,known,self));});
test('open descriptor and named file identities must remain the same',()=>{const root=fs.mkdtempSync('/private/tmp/b018-worker-');try{fs.chmodSync(root,0o700);const file=path.join(root,'input'),data=Buffer.from('{"a":1}');fs.writeFileSync(file,data,{mode:0o600});assert.deepEqual(checkedFile(file,sha(data)),data);const original=fs.openSync;let swapped=false;fs.openSync=function(p,...args){const fd=original.call(fs,p,...args);if(p===file&&!swapped){swapped=true;fs.renameSync(file,file+'.old');fs.writeFileSync(file,data,{mode:0o600});}return fd;};try{assert.throws(()=>checkedFile(file,sha(data)));}finally{fs.openSync=original;}}finally{fs.rmSync(root,{recursive:true,force:true});}});
test('complete inventory admits only named generated outputs and rejects extra entries',()=>{const root=fs.mkdtempSync('/private/tmp/b018-tree-'),expected={};try{for(let i=0;i<1024;i++){const name='f'+i;fs.writeFileSync(path.join(root,name),'x',{mode:0o400});expected[name]={kind:'file',sha256:sha(Buffer.from('x')),bytes:1,executable:false};}fs.mkdirSync(path.join(root,'.next'),{mode:0o700});fs.writeFileSync(path.join(root,'next-env.d.ts'),'generated',{mode:0o600});fs.chmodSync(root,0o500);verifyApplication(root,expected);fs.chmodSync(root,0o700);fs.writeFileSync(path.join(root,'.env'),'unexpected',{mode:0o400});fs.chmodSync(root,0o500);assert.throws(()=>verifyApplication(root,expected));}finally{fs.chmodSync(root,0o700);fs.rmSync(root,{recursive:true,force:true});}});
