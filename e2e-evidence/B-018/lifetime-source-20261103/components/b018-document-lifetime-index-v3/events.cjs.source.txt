const hashRE=/^[a-f0-9]{64}$/;
function valid(value,spec){
  if(Array.isArray(spec))return spec.includes(value);
  if(spec==='hash'||spec==='nullableHash')return (value===null&&spec==='nullableHash')||(typeof value==='string'&&value.length===64&&hashRE.test(value));
  if(spec==='nullableNat')return value===null||valid(value,'nat');
  if(spec==='bool')return typeof value==='boolean';
  if(['nat','pos','ctx','globalCtx','status'].includes(spec)){
    const [min,max]={nat:[0,2**31-1],pos:[1,2**31-1],ctx:[1,6],globalCtx:[0,6],status:[100,599]}[spec];
    return Number.isSafeInteger(value)&&value>=min&&value<=max;
  }
  return value===spec;
}
function clean(type,channel,input){
  if(typeof type!=='string'||typeof channel!=='string')throw Error('invalid-event');
  const definition=Object.hasOwn(definitions,type)&&definitions[type];
  if(!definition||!streams.includes(channel)||(definition.stream!=='ANY'&&definition.stream!==channel)||
    !input||Object.getPrototypeOf(input)!==Object.prototype)throw Error('invalid-event');
  const descriptors=Object.getOwnPropertyDescriptors(input), fields=definition.fields;
  if(Reflect.ownKeys(descriptors).length!==Object.keys(fields).length)throw Error('invalid-event');
  const result={};
  for(const [name,spec] of Object.entries(fields)){
    const descriptor=descriptors[name];
    if(!descriptor||!Object.hasOwn(descriptor,'value')||!valid(descriptor.value,spec))throw Error('invalid-event');
    result[name]=descriptor.value;
  }
  return result;
}
function freeze(x){if(x&&typeof x==='object'&&!Object.isFrozen(x)){Object.freeze(x);Object.values(x).forEach(freeze);}return x;}
