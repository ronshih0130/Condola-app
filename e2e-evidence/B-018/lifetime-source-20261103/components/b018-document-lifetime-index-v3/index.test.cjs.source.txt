function duplicate(e,changes={}){return {...e,fields:{...e.fields,...changes}};}
function incident(ctx,reason){return {channel:'CDP',type:'issue',fields:{ctx,reason}};}
function req(a,ctx,path,method='GET'){return fields(a,'pwOpen',ctx,p=>p.path===path&&p.method===method);}
function cdp(a,ctx,path,method='GET'){return fields(a,'cdpOpen',ctx,p=>p.path===path&&p.method===method);}

test('F03 ordinary enabled POST 303 and GET 200 cancellations remain fatal despite successful sibling and HMR retirement',()=>{
  const h=start(a=>{
    const first=fields(a,'pwOpen',2,x=>x.resource==='Fetch'&&x.method==='GET'&&x.path==='tenant');
    const second=fields(a,'cdpOpen',2,x=>x.resource==='Fetch'&&x.method==='GET'&&x.path==='tenant');
    const sibling=a.filter(e=>(e.channel==='PW'&&e.fields.ctx===2&&e.fields.id===first.id&&['pwOpen','pwCookie','pwResponse','pwEnd'].includes(e.type))||
      (e.channel==='CDP'&&e.fields.ctx===2&&e.fields.id===second.id&&['cdpOpen','extraCookie','responseExtra','cdpResponse','cdpEnd'].includes(e.type))||
      (e.channel==='GUARD'&&e.fields.id===first.id&&e.type.startsWith('proxy'))).map(e=>duplicate(e,{id:e.channel==='CDP'?40002:40001}));
    const mark=find(a,'sameDocument',2);a.splice(a.indexOf(mark),0,...sibling);
    for(const [method,path] of [['POST','login'],['GET','tenant']]){
      const p=fields(a,'pwOpen',2,x=>x.method===method&&x.path===path&&x.resource==='Fetch');

      Object.assign(fields(a,'cdpEnd',2,x=>x.id===c.id&&x.hop===c.hop),{terminal:'failed',canceled:true,error:'aborted'});
    }
  });
  assert.equal(h.cut.status,'FATAL');assert.equal(h.cut.rawFailureEventIds.length,16);
  assert.ok(h.index.journal().some(x=>x.ctx===2&&x.type==='pwResponse'&&x.status===200));assert.ok(h.index.journal().some(x=>x.type==='retire'));
