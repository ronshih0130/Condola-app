 async function closeSlot({ctx}){const w=contexts.get(ctx);if(!w||!w.guardOwned||w.plannedClose)throw Error('close-unavailable');if(w.mode==='enabled')armRetirement(w,'landing');const intentId=++closeIntentId;w.closeIntentId=intentId;put('PARENT','closeIntent',{ctx,intentId});w.plannedClose=true;let closing;try{closing=w.context.close();put('PARENT','closeIssued',{ctx,intentId});}catch{issue('PARENT',ctx,'loss');throw Error('close-unavailable');}await closing;let timer;try{await Promise.race([Promise.all([w.physicalSignal,w.detachSignal]),new Promise((_,reject)=>{timer=setTimeout(()=>reject(Error('close-observation-timeout')),10000);})]);}catch{issue('PARENT',ctx,'unavailable');return {status:'UNAVAILABLE'};}finally{clearTimeout(timer);}await Promise.all([...pending].filter(([,owner])=>owner===ctx).map(([p])=>p));if(!w.physicalEventId||!w.detachEventId||!w.page.isClosed()){issue('PARENT',ctx,'unavailable');return {status:'UNAVAILABLE'};}for(const channelName of ['PW','CDP','OBS'])put('PARENT','closed',{ctx,intentId,physicalEventId:w.physicalEventId,detachEventId:channelName==='CDP'?w.detachEventId:0,channelName,binding:mac('ready',[run,ctx,channelName].join(':'))});w.closed=true;return {status:'CLOSE_OBSERVED'};}
 function afterDrain(){if(drainAttempted)throw Error('drain-already-attempted');drainAttempted=true;if(poisoned||contexts.size!==6||[...contexts.values()].some(w=>!w.closed)||pending.size||!ownedGuard){issue('PARENT',0,'unavailable');return {status:'UNAVAILABLE',financialAggregate:'PENDING'};}const snap=ownedGuard.snapshot();if(!snap.closed||snap.failed||snap.denied||snap.pendingHttp||snap.pendingTunnels||snap.clientSockets||snap.upstreamSockets){issue('PARENT',0,'loss');return {status:'UNAVAILABLE',financialAggregate:'PENDING'};}for(const w of contexts.values())put('PARENT','closed',{ctx:w.ctx,intentId:w.closeIntentId,physicalEventId:w.physicalEventId,detachEventId:0,channelName:'GUARD',binding:mac('ready',[run,w.ctx,'GUARD'].join(':'))});return index.afterDrain();}
 function readLifecycleRecord(id){return records.get(id)||null;}
 function getRecoveryMaterial(){if(contexts.size!==6||[...contexts.values()].some(w=>!w.recovery))throw Error('private-recovery-material-unavailable');return [...contexts.values()].map(w=>w.recovery);}
 function journal(){return index.journal();}
