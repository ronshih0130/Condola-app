'use strict';
const {test}=require('node:test'),assert=require('node:assert/strict'),{before,after,USER,ORG1,ORG2}=require('./checks.cjs');
const key=Buffer.alloc(32,7),first='30000000-0000-4000-8000-000000000001',second='30000000-0000-4000-8000-000000000002';
function fixture(){
 const row=(id,org)=>({id,org_id:org,user_id:USER,is_tenant:true,is_owner:false,is_vendor:false,display_name:'private-fake-marker',created_at:'t0',updated_at:'u0'});
 const a={id:first,org:ORG1,user:USER,tenant:true,owner:false,vendor:false,archived:null,environment:'live',liveOrg:null,createdText:'2026-11-03 10:00:00+00',createdUs:'1000000',full:row(first,ORG1),fullText:'first-before'},b={...a,id:second,org:ORG2,full:row(second,ORG2),fullText:'second-before'};
 const base={authCount:0,activeStaff:false,contacts:[a,b],target:{db:'synthetic',oid:'1'},catalog:[{name:'contacts'}],controls:{all:false},tables:{leases:'unchanged',lease_parties:'unchanged'},contactsBusiness:{count:2,commitment:'b'},auditOwner:'condola',audit:[{id:'1',wire:'abcd',row:{}}]};
 const next=structuredClone(base),b2=next.contacts[1];b2.createdUs='1000001';b2.createdText='2026-11-03 10:00:00.000001+00';b2.full.created_at='t1';b2.full.updated_at='u1';b2.fullText='second-after';next.audit.push({id:'2',wire:'dcba',oldText:'second-before',newText:'second-after',row:{org_id:ORG2,table_name:'contacts',row_id:second,action:'update',actor_user_id:null,actor_role:'condola',reason:null,request_id:null,changed_cols:['created_at'],old_row:structuredClone(b.full),new_row:structuredClone(b2.full)}});

}
test('only exact tied contacts qualify and expected single audit validates',()=>{const {base,next,update}=fixture();assert.equal(before(base).second.id,second);const proof=after(base,next,update,key);assert.equal(proof.secondId,second);assert.equal(proof.auditId,'2');assert.equal(JSON.stringify(proof).includes('private-fake-marker'),false);});
test('non-tied original timestamp and extra contact fail before mutation',()=>{for(const f of [s=>s.contacts[1].createdUs='1000001',s=>s.contacts.push(structuredClone(s.contacts[0]))]){const {base}=fixture();f(base);assert.throws(()=>before(base));}});
test('active staff, session, owner/vendor or nonlive organization fail before mutation',()=>{for(const f of [s=>s.authCount=1,s=>s.activeStaff=true,s=>s.contacts[1].owner=true,s=>s.contacts[1].vendor=true,s=>s.contacts[1].environment='sandbox']){const {base}=fixture();f(base);assert.throws(()=>before(base));}});
test('wrong order, changed first contact or target business data fail readback',()=>{for(const f of [s=>s.contacts[1].createdUs='1000002',s=>s.contacts[0].full.updated_at='changed',s=>s.contacts[1].full.is_tenant=false]){const {base,next,update}=fixture();f(next);assert.throws(()=>after(base,next,update,key));}});
test('financial and relation drift, control drift and target identity drift fail readback',()=>{for(const f of [s=>s.tables.leases='changed',s=>s.tables.lease_parties='changed',s=>s.controls.all=true,s=>s.target.oid='2',s=>s.contactsBusiness.commitment='changed']){const {base,next,update}=fixture();f(next);assert.throws(()=>after(base,next,update,key));}});
test('audit replacement, extra audit, altered event or actor fail',()=>{for(const f of [s=>s.audit[0].wire='beef',s=>s.audit.push({id:'3',wire:'cafe',row:{}}),s=>s.audit[1].row.changed_cols=['updated_at'],s=>s.audit[1].row.actor_user_id=USER,s=>s.audit[1].row.old_row.display_name='tampered']){const {base,next,update}=fixture();f(next);assert.throws(()=>after(base,next,update,key));}});
