const uuid=s=>typeof s==='string'&&/^[a-f0-9]{8}(?:-[a-f0-9]{4}){3}-[a-f0-9]{12}$/.test(s);

const equal=(a,b)=>JSON.stringify(canonical(a))===JSON.stringify(canonical(b));
const tag=(key,domain,data)=>crypto.createHmac('sha256',key).update(JSON.stringify([domain,canonical(data)])).digest('hex');
function demand(v){if(!v)throw Error('fixture-order-unavailable');}
function fingerprint(key,domain,items){const rows=items.map(x=>tag(key,'row:'+domain,x)).sort();return {count:rows.length,commitment:tag(key,'rows:'+domain,rows)};}
