'use strict';
const {test}=require('node:test'),assert=require('node:assert/strict'),crypto=require('node:crypto'),{handshake,createTap}=require('./hmr-protocol.cjs');
const {Receiver}=require('../mode-b-repair-override-prep/node_modules/next/dist/compiled/ws');
function frame(op,body=Buffer.alloc(0),masked=false,fin=true){const h=Buffer.from([(fin?128:0)|op,body.length|(masked?128:0)]);if(!masked)return Buffer.concat([h,body]);const key=Buffer.from([1,2,3,4]),v=Buffer.from(body);for(let i=0;i<v.length;i++)v[i]^=key[i%4];return Buffer.concat([h,key,v]);}
function tap(masked){let closes=[],fail=0;const t=createTap({Receiver,isServer:masked,onClose:c=>closes.push(c),onFailure:()=>fail++});return {t,closes,fail:()=>fail};}
function headers(){const key=Buffer.alloc(16,7).toString('base64'),accept=crypto.createHash('sha1').update(key+'258EAFA5-E914-47DA-95CA-C5AB0DC85B11').digest('base64');return [{httpVersion:'1.1',rawHeaders:['Upgrade','websocket','Connection','Upgrade','Sec-WebSocket-Key',key,'Sec-WebSocket-Version','13','Sec-WebSocket-Extensions','permessage-deflate']},{httpVersion:'1.1',statusCode:101,rawHeaders:['Upgrade','websocket','Connection','Upgrade','Sec-WebSocket-Accept',accept]}];}
test('validated handshake allows offer but not negotiated compression',()=>{const [a,b]=headers();assert.equal(handshake(a,b),true);b.rawHeaders.push('Sec-WebSocket-Extensions','permessage-deflate');assert.throws(()=>handshake(a,b));});
test('bad accept, duplicate key and selected protocol fail closed',()=>{for(const v of ['accept','duplicate','protocol']){const [a,b]=headers();if(v==='accept')b.rawHeaders[5]='bad';if(v==='duplicate')a.rawHeaders.push('Sec-WebSocket-Key','bad');if(v==='protocol')b.rawHeaders.push('Sec-WebSocket-Protocol','hmr');assert.throws(()=>handshake(a,b));}});
test('fragmented data followed by empty close in independent heads preserves buffers',()=>{const x=tap(true),chunk=Buffer.concat([frame(1,Buffer.from('a'),true,false),frame(0,Buffer.from('b'),true,true),frame(8,Buffer.alloc(0),true)]),original=Buffer.from(chunk);assert.equal(x.t.feed(chunk),true);assert.deepEqual(chunk,original);assert.deepEqual(x.closes,[null]);assert.equal(x.t.end(),true);});
test('valid independent bilateral numeric close frames',()=>{const a=tap(true),b=tap(false),code=Buffer.from([3,232]);assert.equal(a.t.feed(frame(8,code,true)),true);assert.equal(b.t.feed(frame(8,code,false)),true);assert.deepEqual(a.closes,[1000]);assert.deepEqual(b.closes,[1000]);});
test('coalesced trailing frame after close is fatal',()=>{const x=tap(false);assert.equal(x.t.feed(Buffer.concat([frame(8),frame(1,Buffer.from('x'))])),false);assert.equal(x.fail(),1);});
test('wrong mask, invalid status and incomplete EOF remain fatal',async()=>{for(const bytes of [frame(8,Buffer.alloc(0),false),frame(8,Buffer.from([3,237]),true),Buffer.from([0x88])]){const x=tap(true);x.t.feed(bytes);x.t.end();await new Promise(resolve=>setImmediate(resolve));assert.equal(x.t.state().failed,true);}});
test('oversized payload rejected without allocation',()=>{const x=tap(false),head=Buffer.from([0x82,127,0,0,0,0,0,0x20,0,0]);assert.equal(x.t.feed(head),false);assert.equal(x.fail(),1);});
test('independent partial heads are joined once, wrong-direction mask is rejected',async()=>{const c=tap(true),s=tap(false),masked=frame(8,Buffer.from([3,232]),true),plain=frame(8,Buffer.from([3,232]),false);assert.equal(c.t.feed(masked.subarray(0,3)),true);assert.equal(c.t.feed(masked.subarray(3)),true);assert.equal(s.t.feed(plain.subarray(0,1)),true);assert.equal(s.t.feed(plain.subarray(1)),true);assert.deepEqual(c.closes,[1000]);assert.deepEqual(s.closes,[1000]);const bad=tap(false);bad.t.feed(masked);await new Promise(resolve=>setImmediate(resolve));assert.equal(bad.t.state().failed,true);});
test('duplicate head replay or further data after close fails',()=>{const x=tap(true),head=frame(8,Buffer.alloc(0),true);assert.equal(x.t.feed(head),true);assert.equal(x.t.feed(head),false);});
