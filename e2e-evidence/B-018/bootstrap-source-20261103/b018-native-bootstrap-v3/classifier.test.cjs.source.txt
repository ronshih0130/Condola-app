    journalSha256:digest('bootstrap-journal'),failures:[],actors:[]};
  for(const [actor,last,persona,landing] of [['primary','006','tenant','/t'],['foreign','005','tenant','/t'],['staff','001','admin','/m']]) {
    const actorId='20000000-0000-4000-8000-000000000'+last, group={actor};
    for(const mode of ['native','enabled']) {

      const documents=['/login',landing].map((route,i)=>{

        const token=digest({context,id,loader,rawUrl:webpack+'?v='+id.length});

        pw.requestId='pw'+i;
        if(block) evidence.failures.push(clone(pw),clone(cdp));
        const hop={...binding,type:'Script',navigation:false,uniqueCorrelationCount:1,
          queryKind:'single-renderer-decimal-v',queryCount:1,pw,cdp};
        const mapping={run,generation,mapSha256:digest(assetMap),origin,path:webpack,context,document:id,frame,loader,

          declarationSha256:declaration.declarationSha256,mapSha256:digest(assetMap),compiledSha256:assetMap.compiledSha256});

          mechanism:'v8-precise-coverage',complete:true,executedBytes:javascriptEnabled?120:0,proofSha256:digest(id+'-coverage')};

          reportOnlyHeaderCount:0,effectiveMetaCount:0,captureSha256:digest(id+'-policy')};
        return {context,id,frame,loader,org:ORG,actorId,origin,mainFrame:true,kind:'Document',status:200,
          path:route,terminal:'finished',inventoryComplete:true,inventorySha256:digest(id+'-inventory'),
          declarations:[declaration],hops:[hop],mapping,coverage,policy};
      });
      const session={actorId,persona,org:ORG,context,currentConsent:true,notExpired:true,naturalClick:true,
        sessionVerified:true,visible:true,finalPath:landing,sessionAlias:context+'-session',proofSha256:digest(context+'-db'),
        loginDocument:documents[0].id,landingDocument:documents[1].id};

      else Object.assign(session,{navigation:[['POST','/login',303],['GET','/auth/enter',307],['GET','/auth/continue',307],
        ['GET',landing,200]],redirectsComplete:true});
      group[mode]={context,javascriptEnabled,session,documents};
    }
    evidence.actors.push(group);
  }
  const postDrain={run,freezeSha256:digest(freeze),bootstrapEvidenceSha256:digest(evidence),bootstrapJournalSha256:evidence.journalSha256,
    complete:true,journalSha256:digest('final-journal'),allFailures:clone(evidence.failures),

    recoveryReceiptSha256:digest('independent-recovery')};
  const x={evidence,postDrain,lock:{freeze}};

  return x;

function seal(x) {
  const freezeSha256=digest(x.lock.freeze);
  x.evidence.freezeSha256=freezeSha256;
  x.postDrain.freezeSha256=freezeSha256;
  x.postDrain.bootstrapEvidenceSha256=digest(x.evidence);
  function receipt(stage,data) {return {stage,run:x.lock.freeze.run,freezeSha256,evidenceSha256:digest(data)};}
  x.lock.bootstrapReceipt=receipt('bootstrap',x.evidence);
  x.lock.finalReceipt=receipt('final',x.postDrain);
}
function docs(x) {return [x.evidence.actors[0].native.documents[0],x.evidence.actors[0].enabled.documents[0]];}
function clean(x) {
  x.evidence.failures=[];x.postDrain.allFailures=[];
  docs(x)[0].declarations=[];docs(x)[0].hops=[];
  x.evidence.assetMap=null;x.lock.freeze.assetMapSha256=null;x.postDrain.mapSha256After=null;
  seal(x);
}
test('native admission requires no financial evidence or final receipt; raw events survive final PASS',()=>{
  const x=build(); const a=native(x.evidence,{freeze:x.lock.freeze,bootstrapReceipt:x.lock.bootstrapReceipt});
  assert.deepEqual(a.qualifiedIds,x.evidence.failures.map(f=>f.id));assert.deepEqual(a.rawFailureIds,a.qualifiedIds);
  assert.equal(a.rawStrict,'FAIL');assert.equal(a.nativeAdmission,'ADMIT');assert.equal(a.aggregate,'PENDING');
  assert.deepEqual(final(x.evidence,x.postDrain,x.lock),{...a,aggregate:'PASS',reason:'fully-recovered'});
});
