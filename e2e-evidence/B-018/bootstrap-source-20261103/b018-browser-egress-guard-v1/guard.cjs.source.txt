function target(raw,upgrade=false){
 if(typeof raw!=='string'||raw.length>16384||raw.includes('#')||/[\x00-\x20\x7f\\]/.test(raw))return null;

  !(u.protocol==='http:'||upgrade&&u.protocol==='ws:'))return null;

 // Do not admit normalized host/port/authority variants.
 if(!raw.startsWith(prefix+'/')||raw!==u.href)return null;

}

function cleanHeaders(headers,websocket=false){

  'forwarded','x-forwarded-host','x-forwarded-proto','x-forwarded-for']);

}
