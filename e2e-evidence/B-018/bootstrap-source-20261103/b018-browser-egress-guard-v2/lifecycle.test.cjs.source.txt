function incoming(h){const req=new h.Stream();req.url='http://127.0.0.1:43156/login';req.method='GET';req.headers={};req.rawHeaders=[];req.socket=new h.Sock();const res=new h.Stream();h.server.handler(req,res);return {req,res,q:h.requests.at(-1)};}
test('G01 completed request body followed by response cancellation is fatal and destroys upstream',async()=>{const h=harness(),g=await h.open(),{req,res,q}=incoming(h);req.emit('end');req.emit('close');res.emit('close');assert.equal(g.snapshot().failed,true);assert.equal(q.destroyed,true);assert.ok(h.logs.some(x=>x.reason==='client-response-incomplete'));});
test('G02 close cannot succeed with pending request before socket assignment and late error remains fatal',async()=>{const h=harness(),g=await h.open(),{q}=incoming(h);const closing=g.close();assert.equal(g.snapshot().closed,false);q.emit('error',Error('late'));await assert.rejects(closing,/egress-quarantine/);assert.equal(g.snapshot().closed,false);assert.equal(g.snapshot().failed,true);});
test('G02 close waits for upstream socket close even when server callback fires',async()=>{const h=harness(),g=await h.open(),{q}=incoming(h),s=new h.Sock();q.emit('socket',s);s.destroy=()=>{s.destroyed=true;};const closing=g.close();assert.equal(g.snapshot().closed,false);await assert.rejects(closing,/egress-quarantine/);assert.equal(g.snapshot().closed,false);assert.equal(g.snapshot().upstreamSockets,1);});
test('G03 connection nominated CSP is rejected rather than removed',async()=>{const h=harness(),g=await h.open(),{res,q}=incoming(h);const reply=new h.Stream();reply.statusCode=200;reply.headers={connection:'content-security-policy','content-security-policy':"default-src 'none'"};reply.rawHeaders=['connection','content-security-policy','content-security-policy',"default-src 'none'"];h.callbacks.at(-1)(reply);assert.equal(g.snapshot().failed,true);assert.equal(reply.destroyed,true);assert.equal(q.destroyed,true);assert.equal(res.destroyed,true);});
test('G03 informational response is fatal, not dropped',async()=>{const h=harness(),g=await h.open(),{q}=incoming(h);q.emit('information',{statusCode:103});assert.equal(g.snapshot().failed,true);assert.ok(h.logs.some(x=>x.reason==='unsupported-informational-response'));});

test('ordinary exchange requires all request and response terminals before close',async()=>{const h=harness(),g=await h.open(),{req,res,q}=incoming(h),reply=new h.Stream();reply.statusCode=200;reply.headers={'content-type':'text/html'};reply.rawHeaders=['content-type','text/html'];h.callbacks.at(-1)(reply);req.emit('end');req.emit('close');q.emit('finish');reply.emit('end');reply.emit('close');res.writableFinished=true;res.emit('finish');res.emit('close');assert.equal(g.snapshot().pendingHttp,1);q.emit('close');assert.equal(g.snapshot().pendingHttp,0);const end=await g.close();assert.equal(end.closed,true);assert.equal(end.failed,false);const x=h.privateEvents.filter(e=>e.exchangeId==='x1');assert.deepEqual(x.filter(e=>e.type!=='transport-terminal').map(e=>e.type),['http-request','http-response','http-terminal']);});
test('HMR close requires request terminal and both peer close events',async()=>{const h=harness(),g=await h.open(),sock=new h.Sock();const req={url:'http://127.0.0.1:43156/_next/webpack-hmr',method:'GET',headers:{upgrade:'websocket',connection:'upgrade'},rawHeaders:['upgrade','websocket','connection','upgrade']};h.server.emit('upgrade',req,sock,Buffer.alloc(0));const q=h.requests.at(-1),peer=new h.Sock();const reply={statusCode:101,headers:{upgrade:'websocket',connection:'upgrade'},rawHeaders:['connection','Upgrade','upgrade','websocket']};q.emit('upgrade',reply,peer,Buffer.alloc(0));q.emit('close');assert.equal(g.snapshot().pendingTunnels,1);sock.destroy();assert.equal(g.snapshot().pendingTunnels,0);const end=await g.close();assert.equal(end.closed,true);assert.equal(end.failed,false);assert.equal(h.privateEvents.at(-1).terminal,'both-peers-closed');});
